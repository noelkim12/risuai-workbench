# HMR Core (빌드 파이프라인 · 해시 매니페스트) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** character/module 아티팩트 루트를 파일을 쓰지 않고 RisuAI 네이티브 JSON + SHA-256 에셋 매니페스트로 빌드하는 `risu-workbench-core` API를 만든다 (HMR 서버가 소비).

**Architecture:** 기존 pack 파이프라인의 인메모리 모델 빌더(`buildCharxFromCanonical`, `buildModuleFromCanonicalDirectory`)를 재사용하고, 그 위에 ① card v3 → risu-native 정의 변환기(순수 domain), ② mtime/size 키 해시 캐시(node), ③ HMR 페이로드 조립기(node)를 얹는다. 스펙: `docs/superpowers/specs/2026-07-09-risuai-hmr-sync-design.md`.

**Tech Stack:** TypeScript strict, vitest 4 (`packages/core/tests/**/*.test.ts`), node:crypto, npm workspaces.

## Global Constraints

- 해시 = **SHA-256 lowercase hex** over raw bytes — RisuAI `hasher`(`risuai-pork/src/ts/parser/parser.svelte.ts:955`)와 동일해야 probe 채택이 성립.
- 에셋 플레이스홀더 = `hmr-asset://<sha256hex>` (정확히 이 prefix).
- 프로토콜 상수: `HMR_PROTOCOL_VERSION = 1`, 포트 대역 `41520–41529`, 연결 문자열 `risu-hmr://127.0.0.1:<port>#k=<token>`.
- 에셋 바이트는 무변형 (재인코딩 금지 — 해시 일치 전제).
- HMR 빌드는 아티팩트 루트에 **어떤 파일도 쓰지 않는다** (watcher 루프 방지).
- 테스트 실행: `npm run test --workspace risu-workbench-core -- tests/<file>.test.ts` (repo 루트에서).
- 기존 코드 컨벤션: 한국어 JSDoc (`함수명 함수.` 형식), `@`→`src` alias는 테스트에서 미사용(상대 경로 import).
- 커밋 메시지 끝에 `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.

export interface HmrAssetEntry {
  hash: string;   // sha256 lowercase hex

/**
 * HMR wire 프로토콜 상수·타입.
 * 워크벤치 HMR 서버와 RisuAI 수신 플러그인이 공유하는 계약.
 * 플러그인은 이 파일을 import할 수 없으므로(샌드박스 단일 번들) 동일 정의를 복제한다 —

export const HMR_ASSET_PLACEHOLDER_PREFIX = 'hmr-asset://';

export const HMR_PORT_RANGE = { start: 41520, end: 41529 } as const;

  hash: string;
  ext: string;
  role: string;
  size: number;
}

export interface HmrHealthResponse {
  app: 'risu-workbench-hmr';
  protocolVersion: number;
  project: { name: string; kind: 'character' | 'module'; stableId: string };
  version: number;
}

export interface HmrWatchResponse {
  version: number;
  definitionChanged: boolean;
  changedAssets: string[];
}

export interface HmrPayloadResponse {
  kind: 'character' | 'module';
  data: Record<string, unknown>;
  assets: HmrAssetEntry[];
}

/**
 * hmrAssetPlaceholder 함수.
 * 페이로드 JSON 내 에셋 참조 자리에 들어갈 플레이스홀더 문자열을 만든다.
 */
export function hmrAssetPlaceholder(hash: string): string {
  return `${HMR_ASSET_PLACEHOLDER_PREFIX}${hash}`;
}

/**
 * buildHmrConnectionString 함수.
 * 워크벤치가 표시/복사에 쓰는 연결 문자열을 만든다 (주소+토큰 단일 문자열).
 */
export function buildHmrConnectionString(port: number, token: string): string {
  return `risu-hmr://127.0.0.1:${port}#k=${token}`;
}

// packages/core/src/domain/hmr/charx-to-risu.ts
import type { CharxV3Envelope } from '../charx/blank-char';

/**
 * card v3 → RisuAI 네이티브 character 정의 변환.
 *
 * risuai-pork/src/ts/characterCards.ts의 v3 import 매핑(:860-1000)과
 * convertCharbook(:1036-1140)을 미러링한다.
 *
 * 의도적 생략: SillyTavern 계열 extension 마이그레이션 분기
 * (useProbability/position/selectiveLogic/delay/match_whole_words).
 * HMR 입력은 항상 워크벤치 pack 산출물이므로 해당 extension이 존재하지 않는다.
 */

interface CardBookEntry {
  keys?: string[];
  secondary_keys?: string[];
  insertion_order?: number;
  name?: string;
  comment?: string;
  content?: string;
  mode?: string;
  constant?: boolean;
  selective?: boolean;
  use_regex?: boolean;
  case_sensitive?: boolean;
  folder?: string;
  extensions?: Record<string, unknown>;
}

interface CharxAssetRef {
  type?: string;
  name?: string;
  ext?: string;
  uri?: string;
}

interface CharbookLike {
  entries?: unknown[];
  recursive_scanning?: boolean;
  scan_depth?: number;
  token_budget?: number;
  extensions?: Record<string, unknown>;
}

/**
 * convertCharbookEntriesToGlobalLore 함수.
 * card v3 character_book entries를 risu loreBook 배열로 변환한다.
 * (risuai-pork characterCards.ts:1119-1133 미러)
 */
export function convertCharbookEntriesToGlobalLore(entries: readonly unknown[]): Record<string, unknown>[] {
  const lorebook: Record<string, unknown>[] = [];
  for (const raw of entries) {
    if (typeof raw !== 'object' || raw === null) continue;
    const book = raw as CardBookEntry;
    const extensions = { ...(book.extensions ?? {}) };
    lorebook.push({
      key: (book.keys ?? []).join(', '),
      secondkey: (book.secondary_keys ?? []).join(', '),
      insertorder: book.insertion_order ?? 0,
      comment: book.name ?? book.comment ?? '',
      content: book.content ?? '',
      mode: book.mode ?? 'normal',
      alwaysActive: book.constant ?? false,
      selective: book.selective ?? false,
      extentions: { ...extensions, risu_case_sensitive: book.case_sensitive ?? false },
      activationPercent: extensions.risu_activationPercent,
      loreCache: extensions.risu_loreCache ?? null,
      useRegex: book.use_regex ?? false,
      folder: book.folder,
    });
  }
  return lorebook;
}

/**
 * convertCharxV3ToRisuDefinition 함수.
 * card v3 envelope를 "HMR가 소유하는 정의 필드"만 담은 risu-native 부분 객체로 변환한다.
 * 사용자 상태(chats/chatPage/chaId)는 절대 포함하지 않는다 — merge 정책의 서버측 보증.
 *
 * @param assetPlaceholders - data.assets 0-based index → 'hmr-asset://<hash>' 매핑.
 *   맵에 없는 index의 에셋은 정의에서 제외된다(바이트 소스가 없는 항목).
 */
export function convertCharxV3ToRisuDefinition(
  envelope: CharxV3Envelope,
  assetPlaceholders: ReadonlyMap<number, string>,
): Record<string, unknown> {
  const data = envelope.data as CharxV3Envelope['data'] & {
    assets?: CharxAssetRef[];
    mes_example?: string;
    character_book?: CharbookLike;
  };

  let image: string | undefined;
  const emotionImages: Array<[string, string]> = [];
  const additionalAssets: Array<[string, string, string]> = [];
  const ccAssets: Array<{ type: string; uri: string; name: string; ext: string }> = [];

  (data.assets ?? []).forEach((asset, index) => {
    const placeholder = assetPlaceholders.get(index);
    if (!placeholder) return;
    const fileName = asset.name ?? `asset_${index}`;
    if (asset.type === 'emotion') {
      emotionImages.push([fileName, placeholder]);
    } else if (asset.type === 'x-risu-asset') {
      additionalAssets.push([fileName, placeholder, asset.ext ?? 'unknown']);
    } else if (asset.type === 'icon' && asset.name === 'main') {
      image = placeholder;
    } else {
      ccAssets.push({ type: asset.type ?? 'asset', uri: placeholder, name: fileName, ext: asset.ext ?? 'unknown' });
    }
  });

  const risuExt = data.extensions?.risuai ?? ({} as CharxV3Envelope['data']['extensions']['risuai']);
  const charbook = data.character_book;

  const definition: Record<string, unknown> = {
    name: data.name ?? '',
    desc: data.description ?? '',
    firstMessage: data.first_mes ?? '',
    exampleMessage: data.mes_example ?? '',
    creatorNotes: data.creator_notes ?? '',
    systemPrompt: data.system_prompt ?? '',
    replaceGlobalNote: data.post_history_instructions ?? '',
    alternateGreetings: data.alternate_greetings ?? [],
    tags: data.tags ?? [],
    creator: data.creator ?? '',
    characterVersion: `${data.character_version ?? ''}`,
    personality: data.personality ?? '',
    scenario: data.scenario ?? '',
    globalLore: charbook?.entries ? convertCharbookEntriesToGlobalLore(charbook.entries) : [],
    customscript: risuExt.customScripts ?? [],
    triggerscript: risuExt.triggerscript ?? [],
    additionalText: risuExt.additionalText ?? '',
    utilityBot: risuExt.utilityBot ?? false,
    emotionImages,
    additionalAssets,
    ccAssets,
  };

  if (image !== undefined) definition.image = image;
  if (risuExt.backgroundHTML !== undefined) definition.backgroundHTML = risuExt.backgroundHTML;
  if (risuExt.defaultVariables !== undefined) definition.defaultVariables = risuExt.defaultVariables;

  if (
    charbook &&
    charbook.recursive_scanning !== undefined &&
    charbook.scan_depth !== undefined &&
    charbook.token_budget !== undefined
  ) {
    definition.loreSettings = {
      tokenBudget: charbook.token_budget,
      scanDepth: charbook.scan_depth,
      recursiveScanning: charbook.recursive_scanning,
      fullWordMatching: charbook.extensions?.risu_fullWordMatching ?? false,
    };
  }

  return definition;
}

// packages/core/src/node/hmr-asset-hash.ts
import { createHash } from 'node:crypto';
import fs from 'node:fs';

/**
 * sha256Hex 함수.
 * RisuAI hasher(parser.svelte.ts:955)와 동일한 SHA-256 lowercase hex를 계산한다.
 * 이 값이 RisuAI 에셋 저장 경로(assets/<hash>.<ext>)와 일치해야 probe 채택이 성립한다.
 */
export function sha256Hex(data: Buffer): string {
  return createHash('sha256').update(data).digest('hex');
}

interface HashCacheEntry {
  mtimeMs: number;
  size: number;
  hash: string;
}

/**
 * AssetHashCache 클래스.
 * 경로+mtimeMs+size 키로 파일 해시를 캐시해, 큰 프로젝트에서
 * 저장→반영 사이의 불필요한 재읽기·재해싱을 없앤다.
 */
export class AssetHashCache {
  private readonly entries = new Map<string, HashCacheEntry>();

  hashFile(fsPath: string): string {
    const stat = fs.statSync(fsPath);
    const hit = this.entries.get(fsPath);
    if (hit && hit.mtimeMs === stat.mtimeMs && hit.size === stat.size) {
      return hit.hash;
    }
    const hash = sha256Hex(fs.readFileSync(fsPath));
    this.entries.set(fsPath, { mtimeMs: stat.mtimeMs, size: stat.size, hash });
    return hash;
  }
}

// packages/core/src/node/hmr-build.ts
import fs from 'node:fs';
import path from 'node:path';
import {
  applyManifestImageSelection,
  buildCharxFromCanonical,
} from '../cli/pack/character/workflow';
import { buildModuleFromCanonicalDirectory } from '../cli/pack/module/workflow';
import { convertCharxV3ToRisuDefinition } from '../domain/hmr/charx-to-risu';
import { hmrAssetPlaceholder, type HmrAssetEntry } from '../domain/hmr/protocol';
import { AssetHashCache } from './hmr-asset-hash';

export type HmrAssetSource = { kind: 'file'; path: string } | { kind: 'buffer'; buffer: Buffer };

export interface HmrBuildResult {
  kind: 'character' | 'module';
  data: Record<string, unknown>;
  assets: HmrAssetEntry[];
  assetSources: Map<string, HmrAssetSource>;
}

interface ManifestAssetEntry {
  index: number;
  type: string;
  name: string;
  ext: string;
  extracted_path: string;
}

/**
 * readManifestEntries 함수.
 * assets/manifest.json을 index 오름차순으로 읽는다.
 * mergeAssetsCanonical / module mergeAssets와 동일한 정렬 규칙 —
 * 이 정렬이 data.assets(또는 module.assets tuples) 배열 index와의 정합을 보장한다.
 */
function readManifestEntries(inRoot: string): ManifestAssetEntry[] {
  const manifestPath = path.join(inRoot, 'assets', 'manifest.json');
  if (!fs.existsSync(manifestPath)) return [];
  const parsed = JSON.parse(fs.readFileSync(manifestPath, 'utf-8')) as { assets?: unknown[] };
  if (!Array.isArray(parsed.assets)) return [];
  return parsed.assets
    .filter((entry): entry is Record<string, unknown> => typeof entry === 'object' && entry !== null)
    .filter((entry) => Number.isFinite(entry.index))
    .sort((left, right) => Number(left.index) - Number(right.index))
    .map((entry) => ({
      index: Number(entry.index),
      type: typeof entry.type === 'string' ? entry.type : 'asset',
      name: typeof entry.name === 'string' ? entry.name : '',
      ext: typeof entry.ext === 'string' ? entry.ext : 'png',
      extracted_path: typeof entry.extracted_path === 'string' ? entry.extracted_path : '',
    }));
}

/**
 * readRisucharImage 함수.
 * `.risuchar` marker의 image 필드를 관대하게 읽는다 (없으면 null).
 */
function readRisucharImage(inRoot: string): string | null {
  try {
    const raw = fs.readFileSync(path.join(inRoot, '.risuchar'), 'utf-8');
    const parsed = JSON.parse(raw) as Record<string, unknown>;
    return typeof parsed.image === 'string' && parsed.image.length > 0 ? parsed.image : null;
  } catch {
    return null;
  }
}

interface HashedAssets {
  placeholders: Map<number, string>;
  entries: HmrAssetEntry[];
  sources: Map<string, HmrAssetSource>;
}

/**
 * hashManifestAssets 함수.
 * manifest 항목들을 해시해 (배열 index → placeholder) 맵과 매니페스트 엔트리를 만든다.
 * 파일이 없는 항목은 건너뛴다 (placeholder 미부여 → 정의에서 제외).
 */
function hashManifestAssets(inRoot: string, cache: AssetHashCache): HashedAssets {
  const placeholders = new Map<number, string>();
  const entries: HmrAssetEntry[] = [];
  const sources = new Map<string, HmrAssetSource>();

  readManifestEntries(inRoot).forEach((entry, arrayIndex) => {
    if (!entry.extracted_path) return;
    const absPath = path.join(inRoot, 'assets', entry.extracted_path);
    if (!fs.existsSync(absPath)) return;
    const hash = cache.hashFile(absPath);
    placeholders.set(arrayIndex, hmrAssetPlaceholder(hash));
    if (!sources.has(hash)) {
      sources.set(hash, { kind: 'file', path: absPath });
      entries.push({
        hash,
        ext: entry.ext,
        role: `${entry.type}:${entry.name}`,
        size: fs.statSync(absPath).size,
      });
    }
  });

  return { placeholders, entries, sources };
}

/**
 * buildHmrCharacterPayload 함수.
 * character 아티팩트 루트를 RisuAI 네이티브 정의 + 에셋 매니페스트로 빌드한다.
 * 디스크에 아무것도 쓰지 않는다.
 */
export function buildHmrCharacterPayload(
  inRoot: string,
  cache: AssetHashCache = new AssetHashCache(),
): HmrBuildResult {
  const charx = buildCharxFromCanonical(inRoot, null, 'none');
  applyManifestImageSelection(charx, inRoot, readRisucharImage(inRoot));
  const { placeholders, entries, sources } = hashManifestAssets(inRoot, cache);
  const data = convertCharxV3ToRisuDefinition(charx, placeholders);
  return { kind: 'character', data, assets: entries, assetSources: sources };
}

 * buildHmrModulePayload 함수.
 * module 아티팩트 루트를 RisuModule JSON + 에셋 매니페스트로 빌드한다.
 * module.assets tuple의 uri를 hmr-asset:// 플레이스홀더로 치환한다.
 * 디스크에 아무것도 쓰지 않는다.
 */
export function buildHmrModulePayload(inRoot: string): HmrBuildResult {
  const { module: moduleObj, assetBuffers } = buildModuleFromCanonicalDirectory(inRoot, {});
  const manifestEntries = readManifestEntries(inRoot);
  const entries: HmrAssetEntry[] = [];
  const sources = new Map<string, HmrAssetSource>();

  const tuples = Array.isArray(moduleObj.assets)
    ? (moduleObj.assets as Array<[string | null, string | null, string | null]>)
    : [];

  tuples.forEach((tuple, index) => {
    const buffer = assetBuffers[index];
    if (!buffer) return;
    const hash = sha256Hex(buffer);
    tuple[1] = hmrAssetPlaceholder(hash);
    if (!sources.has(hash)) {
      sources.set(hash, { kind: 'buffer', buffer });
      entries.push({
        hash,
        ext: manifestEntries[index]?.ext ?? 'png',
        role: `${tuple[2] ?? 'asset'}:${tuple[0] ?? index}`,
        size: buffer.byteLength,
      });
    }
  });

  return { kind: 'module', data: moduleObj, assets: entries, assetSources: sources };
}
